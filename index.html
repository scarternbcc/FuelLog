FuelLog
SETUP
SUPABASE
FuelLog uses Supabase to store your data securely in the cloud. Follow these steps to connect your database.
1
Go to supabase.com and create a free account, then create a new project.
2
In your project, open the SQL Editor and run this setup script:

-- Run this in Supabase SQL Editor
create table vehicles (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users not null,
  nickname text, year text, make text,
  model text, fuel text default 'regular',
  created_at timestamptz default now()
);

create table fillups (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users not null,
  vehicle_id uuid references vehicles(id) on delete cascade,
  date date not null,
  odometer numeric not null,
  gallons numeric not null,
  price numeric, notes text,
  created_at timestamptz default now()
);

alter table vehicles enable row level security;
alter table fillups enable row level security;

create policy "users own vehicles" on vehicles for all using (auth.uid() = user_id) with check (auth.uid() = user_id);
create policy "users own fillups" on fillups for all using (auth.uid() = user_id) with check (auth.uid() = user_id);
3
Go to Project Settings → API and copy your Project URL and anon public key below.
